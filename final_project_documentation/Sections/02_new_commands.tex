% Keep single lines from being stranded at the top or bottom of a page. Safe
% to force hard here because the document is \raggedbottom -- there is no
% column-height stretching to be thrown off by the extra page-breaking
% constraint.
\widowpenalty=10000
\clubpenalty=10000
\brokenpenalty=10000

% Give the line-breaker much more room before it accepts an overfull line.
% This document mixes justified prose with many inline \texttt/\code spans
% (identifiers, paths), which are poor at absorbing interword stretch; a
% generous emergency stretch lets TeX quietly loosen such lines instead of
% letting them bleed into the margin.
\tolerance=9999
\emergencystretch=3em
\hbadness=10000

% ---------------------------------------------------------------------------
% Heading style: technical-documentation look, not a thesis/book look.
% Numbered like "1  Project Overview" / "1.1 Objectives and Scope", no
% "Chapter N" prefix, minimal vertical whitespace, a thin accent rule under
% chapter titles rather than a full blank chapter-opening page.
% ---------------------------------------------------------------------------
\titleformat{\chapter}[hang]
  {\normalfont\Huge\bfseries\color{ThesisNavy}}
  {\thechapter}
  {0.6em}
  {}
\titlespacing*{\chapter}{0pt}{0pt}{18pt}

\titleformat{\section}
  {\normalfont\Large\bfseries\color{ThesisNavy}}
  {\thesection}{0.7em}{}
\titlespacing*{\section}{0pt}{16pt}{6pt}

\titleformat{\subsection}
  {\normalfont\large\bfseries\color{ThesisText}}
  {\thesubsection}{0.6em}{}
\titlespacing*{\subsection}{0pt}{12pt}{4pt}

% Number subsections and show them in the ToC, matching how a reference
% document is actually navigated (readers jump to "3.4 CI/CD Pipeline", not
% to an unnumbered heading they can't cite).
\setcounter{secnumdepth}{3}
\setcounter{tocdepth}{2}

% ---------------------------------------------------------------------------
% Page numbering: "Cloud Interns" running header, "Ejada Cloud 2026" plus a
% zero-padded current/total page count in the footer (the book class forces
% "plain" on every page regardless of \pagestyle elsewhere, so "plain" is
% redefined directly rather than fought). \ifDocArabic switches the counter
% display to plain arabic zero-padding once the front matter's roman-numeral
% pages give way to the numbered body (set true right after
% \pagenumbering{arabic} in main.tex).
% ---------------------------------------------------------------------------
\newif\ifDocArabic
\newcommand{\zeropagenum}{\ifnum\value{page}<10 0\fi\arabic{page}}
\fancypagestyle{plain}{%
  \fancyhf{}
  \renewcommand{\headrulewidth}{0pt}
  \renewcommand{\footrulewidth}{0pt}
  \fancyhead[L]{\small\sffamily\color{black}Cloud Interns}
  \fancyfoot[L]{\small\sffamily\color{black}Ejada Cloud 2026}
  \fancyfoot[R]{\small\sffamily\color{black}\ifDocArabic\zeropagenum\else\thepage\fi\;/\;\pageref{LastPage}}
}
\pagestyle{plain}
